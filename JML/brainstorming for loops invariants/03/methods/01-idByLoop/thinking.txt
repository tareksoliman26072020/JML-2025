WhileCountingUp, because:
1)
loopInitFacts = [ "i == 0" ]
This says the loop starts at a known lower bound.

2) The guard compares i with a stable upper bound
loopGuard = "i < n"
loopBoundStabilityFacts = [ "n is read-only during the loop" ]
loopReadOnlyVars = [ "n" ]

This says the loop continues while i is below n, and n itself is not being changed by the loop.
For idByLoop, n is stable, so n - i is meaningful as a termination measure.

3) i is THE ONLY loop counter (THE STRIDE IS 1)
This distinguishes WhileCountingUp from: WhileTwoFrontier, WhileBinarySearch

4) i has bounds 0 <= i <= n

5) i increases by EXACTLY 1
This distinguishes WhileCountingUp from: WhileStrided

6) The only assigned variable is i
loopAssignments = [ "i" ]
loopFrameTargets = [ "i" ]

there is no (loopAssignments = [ "i", "s" ]), and no (loopFieldAssignments = [ "x" ]),
this means it does not look like a local accumulator loop,
and it is not a field accumulator loop.
And there are no array writes, so it is not an array mutation loop.

7) There are no accumulators, therefore it's not a "WhileAccumulator"

inferCounterPatterns :: LoopSummary -> [(LoopPattern, [LoopSummaryTag])]

inferCounterPattern :: LoopSummary -> String -> [(LoopPattern, [LoopSummaryTag])]
inferCounterPattern summary counterName =
  case lookup counterName (loopCountersDevelopmentTrajectory summary) of

    Just (Increasing step)
      | isOne step ->
          [ ( CounterPat CountingUp
            , counterPatternTags summary counterName
            )
          ]

      | otherwise ->
          [ ( CounterPat StridedCounting
            , counterPatternTags summary counterName
            )
          ]

    Just (Decreasing step)
      | isOne step ->
          [ ( CounterPat CountingDown
            , counterPatternTags summary counterName
            )
          ]

      | otherwise ->
          [ ( CounterPat StridedCounting
            , counterPatternTags summary counterName
            )
          ]

    Just (Mixed _) ->
      [ ( CounterPat ConditionalCounterMovement
        , [ LoopCounters
          , LoopCountersDevelopmentTrajectory
          ]
        )
      ]

    Just ReadOnly ->
      []

    Nothing ->
      []

-- Tags for CounterPat CountingUp
counterPatternTags :: LoopSummary -> String -> [LoopSummaryTag]
counterPatternTags summary counterName =
  concat
    [ [LoopCounters]

        then [LoopAssignments]
        else []

-- Stable bound
inferStableBoundPatterns :: LoopSummary -> [(LoopPattern, [LoopSummaryTag])]
inferStableBoundPatterns summary =
  concatMap checkBound (loopCountersBounds summary)
  where
    checkBound :: (SymExpr, String, SymExpr)
               -> [(LoopPattern, [LoopSummaryTag])]
    checkBound (_lower, counterName, upperBound)
      | counterName `elem` loopCounters summary
        && any (guardIsCounterLtBound counterName upperBound) (loopGuards summary)
        && isReadOnlyBound upperBound summary
        =
          [ ( BoundPat StableBound
            , stableBoundTags summary counterName upperBound
            )
          ]

      | otherwise =
          []

-- Tags for BoundPat StableBound
stableBoundTags :: LoopSummary -> String -> SymExpr -> [LoopSummaryTag]

  concat
    [ [LoopCounters]
    , [LoopCountersBounds]

        then [LoopBoundStabilityFacts]
        else []

    , if isReadOnlyBoundViaReadOnlyVars upperBound summary
        then [LoopReadOnlyVars]
        else []

    , if hasDecreasesOfBoundMinusCounter upperBound counterName summary
        then [LoopDecreasesCandidate]
        else []
    ]

mentionsVar :: String -> SymExpr -> Bool
mentionsVar target expr =
  case expr of
    SymVar _ name ->
      name == target

    SymInt _ ->
      False

    SBin left _op right ->
      mentionsVar target left || mentionsVar target right

    -- Add your other SymExpr constructors here.
    _ ->
      False

guardIsCounterLtBound :: String -> SymExpr -> SymExpr -> Bool
guardIsCounterLtBound counterName upperBound guard =
  case guard of
    SBin left Lt right ->
      symVarName left == Just counterName && right == upperBound

      False
